idf_component_register(
    SRCS "main.c" "c3_store.c" "c3_link.c" "c3_fleet.c" "c3_web.c"
    INCLUDE_DIRS "."
    REQUIRES link_proto fleet_proto espnow_link status_led nvs_flash esp_wifi esp_netif
             esp_event esp_http_server esp_driver_uart esp_timer esp_hw_support mbedtls
    EMBED_FILES "www/index.html")

# The admin password never enters the firmware: only SHA-256(salt ":" password)
# does (admin_hash.h, read by c3_web.c), 32 bytes whatever its length. It is read
# from sdkconfig.json, byte-exact (sdkconfig.cmake would expand a "${...}" in it).
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    set(admin_salt "fpv58-checkpoint/c3")
    set(admin_max 128)   # the Authorization header must fit the 1 KiB header budget
    idf_build_get_property(sdkconfig_json SDKCONFIG_JSON)
    file(READ "${sdkconfig_json}" sdkconfig_text)
    string(JSON admin_pw GET "${sdkconfig_text}" C3_ADMIN_PASSWORD)
    string(LENGTH "${admin_pw}" admin_len)
    if(admin_len GREATER admin_max)
        message(FATAL_ERROR "CONFIG_C3_ADMIN_PASSWORD is longer than ${admin_max} bytes")
    endif()
    if(admin_len EQUAL 0)
        set(admin_open 1)
    else()
        set(admin_open 0)
    endif()
    string(SHA256 admin_hash "${admin_salt}:${admin_pw}")
    string(REGEX REPLACE "(..)" "0x\\1, " admin_hash "${admin_hash}")
    unset(admin_pw)
    unset(sdkconfig_text)
    file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/admin_hash.h" @ONLY CONTENT [=[
/* Generated by main/CMakeLists.txt from CONFIG_C3_ADMIN_PASSWORD; do not edit. */
#pragma once
#define ADMIN_PASSWORD_MAX @admin_max@
#define ADMIN_AUTH_OPEN @admin_open@   /* 1: empty password, admin auth disabled */
#define ADMIN_SALT "@admin_salt@"
#define ADMIN_HASH { @admin_hash@}   /* SHA-256(ADMIN_SALT ":" password) */
]=])
    target_include_directories(${COMPONENT_LIB} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
endif()
